```bash
# 1. 启动 Redis
docker run -d --name redis -p 6379:6379 redis:7-alpine

# 2. 设置环境变量
export OPENAI_API_KEY=your_key

# 3. 启动服务
cd easywing-platform-ai
mvn spring-boot:run

# 4. 运行测试
```
